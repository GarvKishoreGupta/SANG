static void OLED_Command(uint8_t cmd)
{
    uint8_t buf[2] = {0x00, cmd};

    HAL_I2C_Master_Transmit(
        &hi2c1, OLED_ADDR, buf, 2, 100
    );
}

static void OLED_Data(uint8_t data)
{
    uint8_t buf[2] = {0x40, data};

    HAL_I2C_Master_Transmit(
        &hi2c1, OLED_ADDR, buf, 2, 100
    );
}

static void OLED_Init(void)
{
    HAL_Delay(100);

    OLED_Command(0xAE);
    OLED_Command(0xD5); OLED_Command(0x80);
    OLED_Command(0xA8); OLED_Command(0x3F);
    OLED_Command(0xD3); OLED_Command(0x00);
    OLED_Command(0x40);
    OLED_Command(0xAD); OLED_Command(0x8B);
    OLED_Command(0xA1);
    OLED_Command(0xC8);
    OLED_Command(0xDA); OLED_Command(0x12);
    OLED_Command(0x81); OLED_Command(0x7F);
    OLED_Command(0xD9); OLED_Command(0x22);
    OLED_Command(0xDB); OLED_Command(0x35);
    OLED_Command(0xA4);
    OLED_Command(0xA6);
    OLED_Command(0xAF);
}

static void OLED_SetPosition(uint8_t page, uint8_t col)
{
    col += 2;

    OLED_Command(0xB0 | page);
    OLED_Command(0x10 | (col >> 4));
    OLED_Command(0x00 | (col & 0x0F));
}

static void OLED_Clear(void)
{
    for (uint8_t page = 0; page < 8; page++)
    {
        OLED_SetPosition(page, 0);

        for (uint8_t col = 0; col < 128; col++)
            OLED_Data(0x00);
    }
}

static void OLED_Char(char c)
{
    if (c >= 'a' && c <= 'z')
        c -= ('a' - 'A');

    if (c < 32 || c > 90)
        c = ' ';

    uint8_t index = (uint8_t)c - 32;

    for (uint8_t i = 0; i < 5; i++)
        OLED_Data(font5x7[index][i]);

    OLED_Data(0x00);
}

static void OLED_WriteAt(uint8_t page, uint8_t col, const char *str)
{
    OLED_SetPosition(page, col);
    while (*str && col < 122) { OLED_Char(*str++); col = (uint8_t)(col + 6); }
}

static void UpdateSensorCorner(int16_t temperature, uint16_t humidity, uint8_t dhtOK, uint16_t smoke)
{
    char line[24];
    /* Compact top-line values fit the 128-pixel OLED (about 21 characters). */
    if (dhtOK)
        snprintf(line, sizeof(line), "T%d.%d H%u.%u S%u",
                 temperature / 10,
                 (temperature < 0 ? -temperature : temperature) % 10,
                 humidity / 10U, humidity % 10U, smoke);
    else
        snprintf(line, sizeof(line), "T--.- H--.- S%u", smoke);
    OLED_WriteAt(0, 0, line);
}

static void Keypad_Init(void)
{
    GPIO_InitTypeDef gpio = {0};

    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOB_CLK_ENABLE();

    /* Rows PB0-PB3: output push-pull */
    gpio.Pin = GPIO_PIN_0 | GPIO_PIN_1 |
               GPIO_PIN_2 | GPIO_PIN_3;
    gpio.Mode = GPIO_MODE_OUTPUT_PP;
    gpio.Pull = GPIO_NOPULL;
    gpio.Speed = GPIO_SPEED_FREQ_LOW;

    HAL_GPIO_Init(GPIOB, &gpio);

    HAL_GPIO_WritePin(
        GPIOB,
        GPIO_PIN_0 | GPIO_PIN_1 |
        GPIO_PIN_2 | GPIO_PIN_3,
        GPIO_PIN_SET
    );

    /* Column C1: PA15 */
    gpio.Pin = GPIO_PIN_15;
    gpio.Mode = GPIO_MODE_INPUT;
    gpio.Pull = GPIO_PULLUP;
    HAL_GPIO_Init(GPIOA, &gpio);

    /* Columns C2 and C3: PB12 and PB13 */
    gpio.Pin = GPIO_PIN_12 | GPIO_PIN_13;
    gpio.Mode = GPIO_MODE_INPUT;
    gpio.Pull = GPIO_PULLUP;
    HAL_GPIO_Init(GPIOB, &gpio);
}

static char Keypad_GetKey(void)
{
    for (int row = 0; row < 4; row++)
    {
        /* All rows HIGH */
        for (int r = 0; r < 4; r++)
        {
            HAL_GPIO_WritePin(
                rowPort[r], rowPin[r], GPIO_PIN_SET
            );
        }

        /* Current row LOW */
        HAL_GPIO_WritePin(
            rowPort[row], rowPin[row], GPIO_PIN_RESET
        );

        HAL_Delay(2);

        for (int col = 0; col < 3; col++)
        {
            if (HAL_GPIO_ReadPin(
                    colPort[col], colPin[col]
                ) == GPIO_PIN_RESET)
            {
                HAL_Delay(20);

                if (HAL_GPIO_ReadPin(
                        colPort[col], colPin[col]
                    ) == GPIO_PIN_RESET)
                {
                    char key = keyMap[row][col];

                    /* Wait until key is released */
                    while (HAL_GPIO_ReadPin(
                               colPort[col], colPin[col]
                           ) == GPIO_PIN_RESET)
                    {
                        HAL_Delay(5);
                    }

                    HAL_Delay(30);
                    return key;
                }
            }
        }
    }

    return '\0';
}

/* DHT22 microsecond driver copied from the supplied working module. */
static void DHT22_DelayUs(uint32_t us)
{
    uint32_t start = DWT->CYCCNT;
    uint32_t cycles = (SystemCoreClock / 1000000U) * us;

    while ((uint32_t)(DWT->CYCCNT - start) < cycles)
    {
    }
}

static void DHT22_SetInput(void)

    gpio.Pin = DHT22_PIN;
    gpio.Mode = GPIO_MODE_INPUT;
    gpio.Pull = GPIO_PULLUP;
    gpio.Speed = GPIO_SPEED_FREQ_LOW;
    HAL_GPIO_Init(DHT22_PORT, &gpio);
}

static void DHT22_SetOutput(void)
{
    GPIO_InitTypeDef gpio = {0};
    gpio.Pin = DHT22_PIN;
    gpio.Mode = GPIO_MODE_OUTPUT_OD;
    gpio.Pull = GPIO_PULLUP;
    gpio.Speed = GPIO_SPEED_FREQ_LOW;
    HAL_GPIO_Init(DHT22_PORT, &gpio);
}

static void DHT22_Init(void)
{
    GPIO_InitTypeDef gpio = {0};

    __HAL_RCC_GPIOB_CLK_ENABLE();

    CoreDebug->DEMCR |= CoreDebug_DEMCR_TRCENA_Msk;
    DWT->CYCCNT = 0;
    DWT->CTRL |= DWT_CTRL_CYCCNTENA_Msk;

    gpio.Pin = DHT22_PIN;
    gpio.Mode = GPIO_MODE_OUTPUT_OD;
    gpio.Pull = GPIO_PULLUP;
    gpio.Speed = GPIO_SPEED_FREQ_LOW;
    HAL_GPIO_Init(DHT22_PORT, &gpio);

    HAL_GPIO_WritePin(DHT22_PORT, DHT22_PIN, GPIO_PIN_SET);
}

static uint8_t DHT22_WaitLevel(GPIO_PinState level,
                               uint32_t timeout_us)
{
    uint32_t start = DWT->CYCCNT;
    uint32_t timeout_cycles =
        (SystemCoreClock / 1000000U) * timeout_us;

    while (HAL_GPIO_ReadPin(DHT22_PORT, DHT22_PIN) != level)
    {
        if ((uint32_t)(DWT->CYCCNT - start) >= timeout_cycles)
            return 0;
    }

    return 1;
}

static uint8_t DHT22_ReadByte(uint8_t *value)
{
    uint8_t result = 0;

    for (uint8_t i = 0; i < 8; i++)
    {
        if (!DHT22_WaitLevel(GPIO_PIN_RESET, 100))
            return 0;

        if (!DHT22_WaitLevel(GPIO_PIN_SET, 100))
            return 0;

        DHT22_DelayUs(40);
        result <<= 1;

        if (HAL_GPIO_ReadPin(DHT22_PORT, DHT22_PIN)
            == GPIO_PIN_SET)
        {
            result |= 1;
        }
    }

    *value = result;
    return 1;
}

static uint8_t DHT22_Read(int16_t *temperature,
                          uint16_t *humidity)
{
    uint8_t data[5] = {0};

    DHT22_SetOutput();
    HAL_GPIO_WritePin(DHT22_PORT, DHT22_PIN, GPIO_PIN_RESET);
    HAL_Delay(2);

    HAL_GPIO_WritePin(DHT22_PORT, DHT22_PIN, GPIO_PIN_SET);
    DHT22_DelayUs(30);
    DHT22_SetInput();

    if (!DHT22_WaitLevel(GPIO_PIN_RESET, 100))
        return 0;

    if (!DHT22_WaitLevel(GPIO_PIN_SET, 100))
        return 0;

    if (!DHT22_WaitLevel(GPIO_PIN_RESET, 100))
        return 0;

    for (uint8_t i = 0; i < 5; i++)
    {
        if (!DHT22_ReadByte(&data[i]))
            return 0;
    }

    if ((uint8_t)(data[0] + data[1] + data[2] + data[3])
        != data[4])
    {
        return 0;
    }

    *humidity = ((uint16_t)data[0] << 8) | data[1];

    uint16_t rawTemperature =
        ((uint16_t)(data[2] & 0x7F) << 8) | data[3];

    if (data[2] & 0x80)
        *temperature = -(int16_t)rawTemperature;
    else
        *temperature = (int16_t)rawTemperature;

    return 1;
}


static uint16_t MQ2_ReadADC(void)
{
    uint16_t value = 0;
    if (HAL_ADC_Start(&hadc1) != HAL_OK) return 0;
    if (HAL_ADC_PollForConversion(&hadc1, 20) == HAL_OK) value = (uint16_t)HAL_ADC_GetValue(&hadc1);
    HAL_ADC_Stop(&hadc1);
    return value;
}

static void Buzzer_Tone(uint8_t enable)
{
    if (!enable) { HAL_GPIO_WritePin(BUZZER_PORT, BUZZER_PIN, GPIO_PIN_RESET); return; }
    for (uint16_t i = 0; i < 100; i++) {
        HAL_GPIO_WritePin(BUZZER_PORT, BUZZER_PIN, GPIO_PIN_SET); HAL_Delay(1);
        HAL_GPIO_WritePin(BUZZER_PORT, BUZZER_PIN, GPIO_PIN_RESET); HAL_Delay(1);
    }
}

/* RC522 SPI driver copied from the supplied RFID module. */
static void RFID_Select(void)
{
    HAL_GPIO_WritePin(RFID_CS_PORT, RFID_CS_PIN, GPIO_PIN_RESET);
}

static void RFID_Deselect(void)
{
    HAL_GPIO_WritePin(RFID_CS_PORT, RFID_CS_PIN, GPIO_PIN_SET);
}

void RFID_WriteRegister(uint8_t reg, uint8_t value)
{
    uint8_t tx[2];

    tx[0] = (uint8_t)((reg << 1) & 0x7E);
    tx[1] = value;

    RFID_Select();
    HAL_SPI_Transmit(&hspi1, tx, 2, HAL_MAX_DELAY);
    RFID_Deselect();
}

uint8_t RFID_ReadRegister(uint8_t reg)
{
    uint8_t tx[2] = {
        (uint8_t)(((reg << 1) & 0x7E) | 0x80), 0
    };
    uint8_t rx[2] = {0, 0};

    RFID_Select();
    HAL_SPI_TransmitReceive(&hspi1, tx, rx, 2, HAL_MAX_DELAY);
    RFID_Deselect();

    return rx[1];
}

static void RFID_SetBits(uint8_t reg, uint8_t mask)
{
    RFID_WriteRegister(reg, RFID_ReadRegister(reg) | mask);
}

static void RFID_ClearBits(uint8_t reg, uint8_t mask)
{
    RFID_WriteRegister(reg,
                       RFID_ReadRegister(reg) & (uint8_t)(~mask));
}

static void RFID_AntennaOn(void)
{
    if ((RFID_ReadRegister(REG_TX_CONTROL) & 0x03) != 0x03)
        RFID_SetBits(REG_TX_CONTROL, 0x03);
}

void RFID_Init(void)
{
    HAL_GPIO_WritePin(RFID_RST_PORT, RFID_RST_PIN, GPIO_PIN_SET);
    HAL_Delay(50);

    RFID_WriteRegister(REG_COMMAND, CMD_SOFT_RESET);
    HAL_Delay(50);

    RFID_WriteRegister(REG_T_MODE, 0x8D);
    RFID_WriteRegister(REG_T_PRESCALER, 0x3E);
    RFID_WriteRegister(REG_T_RELOAD_H, 0x00);
    RFID_WriteRegister(REG_T_RELOAD_L, 30);
    RFID_WriteRegister(REG_TX_ASK, 0x40);
    RFID_WriteRegister(REG_MODE, 0x3D);

    RFID_AntennaOn();
}

static uint8_t RFID_Transceive(uint8_t *sendData,
                              uint8_t sendLen,
                              uint8_t *backData,
                              uint8_t *backLen,
                              uint8_t validBits)
{
    uint8_t irq;
    uint8_t error;
    uint32_t timeout = 20000;

    RFID_WriteRegister(REG_COMMAND, CMD_IDLE);
    RFID_WriteRegister(REG_COM_IRQ, 0x7F);
    RFID_WriteRegister(REG_FIFO_LEVEL, 0x80);

    for (uint8_t i = 0; i < sendLen; i++)
        RFID_WriteRegister(REG_FIFO_DATA, sendData[i]);

    RFID_WriteRegister(REG_BIT_FRAMING, validBits);
    RFID_WriteRegister(REG_COMMAND, CMD_TRANSCEIVE);
    RFID_SetBits(REG_BIT_FRAMING, 0x80);

    do
    {
        irq = RFID_ReadRegister(REG_COM_IRQ);
        timeout--;
    }
    while (timeout && !(irq & 0x30) && !(irq & 0x01));

    RFID_ClearBits(REG_BIT_FRAMING, 0x80);

    if (timeout == 0 || (irq & 0x01))
        return 0;

    error = RFID_ReadRegister(REG_ERROR);

    if (error & 0x1B)
        return 0;

    if (backData && backLen)
    {
        uint8_t fifoCount = RFID_ReadRegister(REG_FIFO_LEVEL);

        if (fifoCount > *backLen)
            return 0;

        *backLen = fifoCount;

        for (uint8_t i = 0; i < fifoCount; i++)
            backData[i] = RFID_ReadRegister(REG_FIFO_DATA);
    }

    return 1;
}

uint8_t RFID_ReadUID(uint8_t *uid)
{
    uint8_t request = PICC_REQA;
    uint8_t atqa[2];
    uint8_t atqaLen = sizeof(atqa);

    if (!RFID_Transceive(&request, 1, atqa, &atqaLen, 0x07))
        return 0;

    uint8_t anticoll[2] = {PICC_ANTICOLL, 0x20};
    uint8_t response[5];
    uint8_t responseLen = sizeof(response);

    if (!RFID_Transceive(anticoll, 2, response,
                         &responseLen, 0x00))
        return 0;

    if (responseLen != 5)
        return 0;

    if ((uint8_t)(response[0] ^ response[1] ^
                  response[2] ^ response[3]) != response[4])
        return 0;

    for (uint8_t i = 0; i < 4; i++)
        uid[i] = response[i];

    return 1;
}


void GarageServo_Set(uint16_t pulse)
{
    __HAL_TIM_SET_COMPARE(&htim1, TIM_CHANNEL_2, pulse);
}

/* Combined GPIO initialization. Avoid initializing pins owned by alternate functions here. */
static void MX_GPIO_Init(void)
{
    GPIO_InitTypeDef g = {0};
    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOB_CLK_ENABLE();

    /* Initial output levels */
    HAL_GPIO_WritePin(GPIOA, GPIO_PIN_4, GPIO_PIN_SET); /* RC522 CS idle */
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_10, GPIO_PIN_SET); /* RC522 reset */
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_8 | GPIO_PIN_9, GPIO_PIN_RESET); /* LED/buzzer */
    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_0 | GPIO_PIN_1 | GPIO_PIN_2 | GPIO_PIN_3, GPIO_PIN_SET);

    /* RC522 chip select and reset */
    g.Pin = GPIO_PIN_4; g.Mode = GPIO_MODE_OUTPUT_PP; g.Pull = GPIO_NOPULL; g.Speed = GPIO_SPEED_FREQ_VERY_HIGH;
    HAL_GPIO_Init(GPIOA, &g);
    g.Pin = GPIO_PIN_10; g.Speed = GPIO_SPEED_FREQ_LOW; HAL_GPIO_Init(GPIOB, &g);

    /* OLED I2C1 PB6/PB7 */
    g.Pin = GPIO_PIN_6 | GPIO_PIN_7; g.Mode = GPIO_MODE_AF_OD; g.Pull = GPIO_PULLUP;
    g.Speed = GPIO_SPEED_FREQ_VERY_HIGH; g.Alternate = GPIO_AF4_I2C1; HAL_GPIO_Init(GPIOB, &g);
    /* RC522 SPI1 PA5/PA6/PA7 */

    g.Speed = GPIO_SPEED_FREQ_VERY_HIGH; g.Alternate = GPIO_AF5_SPI1; HAL_GPIO_Init(GPIOA, &g);

    /* MQ-2 analog PA0 */
    g.Pin = GPIO_PIN_0; g.Mode = GPIO_MODE_ANALOG; g.Pull = GPIO_NOPULL; HAL_GPIO_Init(GPIOA, &g);
    /* PIR PA1 */
    g.Pin = GPIO_PIN_1; g.Mode = GPIO_MODE_INPUT; g.Pull = GPIO_NOPULL; HAL_GPIO_Init(GPIOA, &g);
    /* Main door servo PA8 / TIM1_CH1, garage servo PA9 / TIM1_CH2 */
    g.Pin = GPIO_PIN_8 | GPIO_PIN_9; g.Mode = GPIO_MODE_AF_PP; g.Pull = GPIO_NOPULL;

    HAL_GPIO_WritePin(GPIOB, GPIO_PIN_0 | GPIO_PIN_1 | GPIO_PIN_2 | GPIO_PIN_3, GPIO_PIN_SET);
    /* Keypad columns PB12/PB13 */
    g.Pin = GPIO_PIN_12 | GPIO_PIN_13; g.Mode = GPIO_MODE_INPUT; g.Pull = GPIO_PULLUP; HAL_GPIO_Init(GPIOB, &g);
    /* DHT22 PB14 is configured dynamically by its driver */
    /* LDR PB15 */
    g.Pin = GPIO_PIN_15; g.Mode = GPIO_MODE_INPUT; g.Pull = GPIO_PULLDOWN; HAL_GPIO_Init(GPIOB, &g);
    /* LED PB8, buzzer PB9 */
    g.Pin = GPIO_PIN_8 | GPIO_PIN_9; g.Mode = GPIO_MODE_OUTPUT_PP; g.Pull = GPIO_NOPULL; g.Speed = GPIO_SPEED_FREQ_LOW; HAL_GPIO_Init(GPIOB, &g);

int main(void)
{
    HAL_Init();
    SystemClock_Config();
    MX_GPIO_Init();
    MX_I2C1_Init();
    MX_SPI1_Init();
    MX_ADC1_Init();
    MX_TIM1_Init();
    MX_TIM3_Init();

    if (HAL_TIM_PWM_Start(&htim1, TIM_CHANNEL_1) != HAL_OK) Error_Handler();
    if (HAL_TIM_PWM_Start(&htim1, TIM_CHANNEL_2) != HAL_OK) Error_Handler();
    if (HAL_TIM_PWM_Start(&htim3, TIM_CHANNEL_1) != HAL_OK) Error_Handler();
    Servo_SetPosition(SERVO_LOCKED_PULSE);
    GarageServo_Set(GARAGE_SERVO_CLOSE);
    Fan_SetSpeed(FAN_OFF);
    Buzzer_Tone(0);
    DHT22_Init();
    Keypad_Init();
    OLED_Init();
    OLED_Clear();

    RFID_Init();
    uint8_t rfidVersion = RFID_ReadRegister(REG_VERSION);
    uint8_t rfidOK = (rfidVersion != 0x00 && rfidVersion != 0xFF);

    char entered[MAX_PASSWORD_LENGTH + 1] = {0};
    char masked[MAX_PASSWORD_LENGTH + 1] = {0};
    uint8_t length = 0;
    char status[22] = "READY";
    uint32_t lastSensorTick = HAL_GetTick();
    uint32_t lastDHTTick = 0;
    uint32_t lastGasBeepTick = 0;
    uint32_t lastRFIDTick = 0;
    uint32_t lastDoorUnlockTick = 0;
    uint32_t garageOpenUntil = 0;
    uint8_t garageIsOpen = 0;
    uint8_t doorUnlocked = 0;
    uint8_t dhtOK = 0;
    int16_t temperature = 0;
    uint16_t humidity = 0;
    uint16_t smoke = 0;
    uint8_t previousUID[4] = {0};
    uint8_t havePreviousUID = 0;

    UpdatePasswordArea(masked, "READY");
    UpdateSensorCorner(temperature, humidity, 0, smoke);

    /* PIR startup stabilization as in the supplied working module. */
    uint32_t pirStartup = HAL_GetTick();
    while ((HAL_GetTick() - pirStartup) < 30000U) {
        uint32_t now = HAL_GetTick();
        if (now - lastSensorTick >= 500U) {
            smoke = MQ2_ReadADC();
            UpdateSensorCorner(temperature, humidity, dhtOK, smoke);
            lastSensorTick = now;
        }
        HAL_Delay(10);
    }

    while (1)
    {
        uint32_t now = HAL_GetTick();

        /* Update DHT22 and fan about every 2 seconds. */
        if (now - lastDHTTick >= 2000U) {
            int16_t newT = 0; uint16_t newH = 0;
            dhtOK = DHT22_Read(&newT, &newH);
            if (dhtOK) {
                temperature = newT; humidity = newH;
                uint16_t duty = FAN_OFF;
                if (temperature >= (int16_t)TEMP_LIMIT && humidity >= HUM_LIMIT) {
                    if (temperature >= 310 && humidity >= 900) duty = FAN_HIGH;
                    else if (temperature >= 280 && humidity >= 850) duty = FAN_MEDIUM;
                    else duty = FAN_LOW;
                }
                Fan_SetSpeed(duty);
            } else {
                /* Keep last fan setting if a single DHT read fails. */
            }
            lastDHTTick = now;
        }

        /* MQ-2 and the compact OLED corner reading. */
        if (now - lastSensorTick >= 500U) {
            smoke = MQ2_ReadADC();
            UpdateSensorCorner(temperature, humidity, dhtOK, smoke);
            if (smoke >= MQ2_THRESHOLD && (now - lastGasBeepTick >= 1000U)) {
                Buzzer_Tone(1);
                lastGasBeepTick = HAL_GetTick();
            } else if (smoke < MQ2_THRESHOLD) {
                Buzzer_Tone(0);
            }
            lastSensorTick = HAL_GetTick();
        }

        /* Password keypad controls the main door servo on PA8. */
        char key = Keypad_GetKey();
        if (key != '\0') {
            if (key >= '0' && key <= '9') {
                if (length < MAX_PASSWORD_LENGTH) {
                    entered[length] = key; length++; entered[length] = '\0';
                    masked[length - 1] = '*'; masked[length] = '\0';
                    UpdatePasswordArea(masked, "TYPE # TO CHECK");
                }
            } else if (key == '*') {
                memset(entered, 0, sizeof(entered)); memset(masked, 0, sizeof(masked)); length = 0;
                UpdatePasswordArea(masked, "CLEARED");
            } else if (key == '#') {
                if (strcmp(entered, PASSWORD) == 0) {
                    Servo_SetPosition(SERVO_UNLOCKED_PULSE);
                    doorUnlocked = 1; lastDoorUnlockTick = HAL_GetTick();
                    snprintf(status, sizeof(status), "ACCESS GRANTED");
                } else {
                    snprintf(status, sizeof(status), "ACCESS DENIED");
                }
                memset(entered, 0, sizeof(entered)); memset(masked, 0, sizeof(masked)); length = 0;
                UpdatePasswordArea(masked, status);
            }
        }
        if (doorUnlocked && (HAL_GetTick() - lastDoorUnlockTick >= 3000U)) {
            Servo_SetPosition(SERVO_LOCKED_PULSE);
            doorUnlocked = 0;
            UpdatePasswordArea(masked, "DOOR LOCKED");
        }

        /* PIR + LDR: LED on only if motion is detected AND it is dark.
           Assumption retained from supplied code: LDR D0 HIGH means dark. */
        GPIO_PinState ldr = HAL_GPIO_ReadPin(GPIOB, GPIO_PIN_15);
        GPIO_PinState pir = HAL_GPIO_ReadPin(GPIOA, GPIO_PIN_1);
        HAL_GPIO_WritePin(GPIOB, GPIO_PIN_8,
            (ldr == GPIO_PIN_SET && pir == GPIO_PIN_SET) ? GPIO_PIN_SET : GPIO_PIN_RESET);

        /* Close garage after its timed-open interval without blocking other modules. */
        if (garageIsOpen && (int32_t)(HAL_GetTick() - garageOpenUntil) >= 0) {
            GarageServo_Set(GARAGE_SERVO_CLOSE);
            garageIsOpen = 0;
        }

        /* RFID garage control. Do not update OLED from this subsystem.
           Authorized UID opens to the tested 2000 us position for 10 seconds;
           then the existing timeout logic returns it to the tested 1000 us position. */
        if (rfidOK && (HAL_GetTick() - lastRFIDTick >= 100U)) {
            uint8_t uid[4];
            if (RFID_ReadUID(uid)) {
                uint8_t same = havePreviousUID && memcmp(uid, previousUID, 4) == 0;
                if (!same) {
                    memcpy(previousUID, uid, 4); havePreviousUID = 1;
                    if (uid[0] == RFID_KEY_UID_0 && uid[1] == RFID_KEY_UID_1 && uid[2] == RFID_KEY_UID_2 && uid[3] == RFID_KEY_UID_3) {
                        GarageServo_Set(GARAGE_SERVO_OPEN);
                        garageIsOpen = 1;
                        garageOpenUntil = HAL_GetTick() + GARAGE_OPEN_TIME;
                    } else if (uid[0] == RFID_CARD_UID_0 && uid[1] == RFID_CARD_UID_1 && uid[2] == RFID_CARD_UID_2 && uid[3] == RFID_CARD_UID_3) {
                        /* Known unauthorized card: leave garage closed. */
                    }
                }
            } else {
                havePreviousUID = 0;
            }
            lastRFIDTick = HAL_GetTick();
        }
        HAL_Delay(10);
    }
}

void SystemClock_Config(void)
{
    RCC_OscInitTypeDef osc = {0}; RCC_ClkInitTypeDef clk = {0};
    __HAL_RCC_PWR_CLK_ENABLE();
    __HAL_PWR_VOLTAGESCALING_CONFIG(PWR_REGULATOR_VOLTAGE_SCALE2);
    osc.OscillatorType = RCC_OSCILLATORTYPE_HSI; osc.HSIState = RCC_HSI_ON;
    osc.HSICalibrationValue = RCC_HSICALIBRATION_DEFAULT; osc.PLL.PLLState = RCC_PLL_NONE;
    if (HAL_RCC_OscConfig(&osc) != HAL_OK) Error_Handler();
    clk.ClockType = RCC_CLOCKTYPE_HCLK | RCC_CLOCKTYPE_SYSCLK | RCC_CLOCKTYPE_PCLK1 | RCC_CLOCKTYPE_PCLK2;
    clk.SYSCLKSource = RCC_SYSCLKSOURCE_HSI; clk.AHBCLKDivider = RCC_SYSCLK_DIV1;
    clk.APB1CLKDivider = RCC_HCLK_DIV1; clk.APB2CLKDivider = RCC_HCLK_DIV1;
    if (HAL_RCC_ClockConfig(&clk, FLASH_LATENCY_0) != HAL_OK) Error_Handler();
}

void Error_Handler(void)
{
    __disable_irq();
    while (1) { }
}
