/* =========================================================
   OLED DRIVER
   ========================================================= */
static void OLED_Command(uint8_t cmd)
{
    uint8_t data[2] = {0x00, cmd};
    HAL_I2C_Master_Transmit(&hi2c1, 0x3C << 1,
                            data, 2, HAL_MAX_DELAY);
}

static void OLED_Data(uint8_t value)
{
    uint8_t data[2] = {0x40, value};
    HAL_I2C_Master_Transmit(&hi2c1, 0x3C << 1,
                            data, 2, HAL_MAX_DELAY);
}

static void OLED_SetPosition(uint8_t page, uint8_t column)
{
    OLED_Command(0xB0 | (page & 0x07));
    OLED_Command(column & 0x0F);
    OLED_Command(0x10 | ((column >> 4) & 0x0F));
}

static void OLED_Char(char c)
{
    uint8_t index = 0;

    if (c >= ' ' && c <= 'Z')
        index = (uint8_t)(c - ' ');

    for (uint8_t i = 0; i < 5; i++)
        OLED_Data(font[index][i]);

    OLED_Data(0x00);
}

void OLED_String(uint8_t page, uint8_t column, const char *text)
{
    OLED_SetPosition(page, column);

    while (*text && column <= 122)
    {
        OLED_Char(*text++);
        column += 6;
    }
}

void OLED_Clear(void)
{
    for (uint8_t page = 0; page < 8; page++)
    {
        OLED_SetPosition(page, 0);

        for (uint8_t col = 0; col < 128; col++)
            OLED_Data(0x00);
    }
}

void OLED_Init(void)
{
    HAL_Delay(100);

    OLED_Command(0xAE);
    OLED_Command(0xD5); OLED_Command(0x80);
    OLED_Command(0xA8); OLED_Command(0x3F);
    OLED_Command(0xD3); OLED_Command(0x00);
    OLED_Command(0x40);
    OLED_Command(0xAD); OLED_Command(0x8B);
    OLED_Command(0xA1);
    OLED_Command(0xC8);
    OLED_Command(0xDA); OLED_Command(0x12);
    OLED_Command(0x81); OLED_Command(0x7F);
    OLED_Command(0xD9); OLED_Command(0x22);
    OLED_Command(0xDB); OLED_Command(0x35);
    OLED_Command(0xA4);
    OLED_Command(0xA6);
    OLED_Command(0xAF);

    OLED_Clear();
}

void DisplayStatus(const char *line1, const char *line2)
{
    OLED_Clear();
    OLED_String(1, 0, line1);
    OLED_String(3, 0, line2);
}

/* =========================================================
   RC522 SPI DRIVER
   ========================================================= */
static void RFID_Select(void)
{
    HAL_GPIO_WritePin(RFID_CS_PORT, RFID_CS_PIN, GPIO_PIN_RESET);
}

static void RFID_Deselect(void)
{
    HAL_GPIO_WritePin(RFID_CS_PORT, RFID_CS_PIN, GPIO_PIN_SET);
}

void RFID_WriteRegister(uint8_t reg, uint8_t value)
{
    uint8_t tx[2];

    tx[0] = (uint8_t)((reg << 1) & 0x7E);
    tx[1] = value;

    RFID_Select();
    HAL_SPI_Transmit(&hspi1, tx, 2, HAL_MAX_DELAY);
    RFID_Deselect();
}

uint8_t RFID_ReadRegister(uint8_t reg)
{
    uint8_t tx[2] = {
        (uint8_t)(((reg << 1) & 0x7E) | 0x80), 0
    };
    uint8_t rx[2] = {0, 0};

    RFID_Select();
    HAL_SPI_TransmitReceive(&hspi1, tx, rx, 2, HAL_MAX_DELAY);
    RFID_Deselect();

    return rx[1];
}

static void RFID_SetBits(uint8_t reg, uint8_t mask)
{
    RFID_WriteRegister(reg, RFID_ReadRegister(reg) | mask);
}

static void RFID_ClearBits(uint8_t reg, uint8_t mask)
{
    RFID_WriteRegister(reg,
                       RFID_ReadRegister(reg) & (uint8_t)(~mask));
}

static void RFID_AntennaOn(void)
{
    if ((RFID_ReadRegister(REG_TX_CONTROL) & 0x03) != 0x03)
        RFID_SetBits(REG_TX_CONTROL, 0x03);
}

void RFID_Init(void)
{
    HAL_GPIO_WritePin(RFID_RST_PORT, RFID_RST_PIN, GPIO_PIN_SET);
    HAL_Delay(50);

    RFID_WriteRegister(REG_COMMAND, CMD_SOFT_RESET);
    HAL_Delay(50);

    RFID_WriteRegister(REG_T_MODE, 0x8D);
    RFID_WriteRegister(REG_T_PRESCALER, 0x3E);
    RFID_WriteRegister(REG_T_RELOAD_H, 0x00);
    RFID_WriteRegister(REG_T_RELOAD_L, 30);
    RFID_WriteRegister(REG_TX_ASK, 0x40);
    RFID_WriteRegister(REG_MODE, 0x3D);

    RFID_AntennaOn();
}

static uint8_t RFID_Transceive(uint8_t *sendData,
                              uint8_t sendLen,
                              uint8_t *backData,
                              uint8_t *backLen,
                              uint8_t validBits)
{
    uint8_t irq;
    uint8_t error;
    uint32_t timeout = 20000;

    RFID_WriteRegister(REG_COMMAND, CMD_IDLE);
    RFID_WriteRegister(REG_COM_IRQ, 0x7F);
    RFID_WriteRegister(REG_FIFO_LEVEL, 0x80);

    for (uint8_t i = 0; i < sendLen; i++)
        RFID_WriteRegister(REG_FIFO_DATA, sendData[i]);

    RFID_WriteRegister(REG_BIT_FRAMING, validBits);
    RFID_WriteRegister(REG_COMMAND, CMD_TRANSCEIVE);
    RFID_SetBits(REG_BIT_FRAMING, 0x80);

    do
    {
        irq = RFID_ReadRegister(REG_COM_IRQ);
        timeout--;
    }
    while (timeout && !(irq & 0x30) && !(irq & 0x01));

    RFID_ClearBits(REG_BIT_FRAMING, 0x80);

    if (timeout == 0 || (irq & 0x01))
        return 0;

    error = RFID_ReadRegister(REG_ERROR);

    if (error & 0x1B)
        return 0;

    if (backData && backLen)
    {
        uint8_t fifoCount = RFID_ReadRegister(REG_FIFO_LEVEL);

        if (fifoCount > *backLen)
            return 0;

        *backLen = fifoCount;

        for (uint8_t i = 0; i < fifoCount; i++)
            backData[i] = RFID_ReadRegister(REG_FIFO_DATA);
    }

    return 1;
}

uint8_t RFID_ReadUID(uint8_t *uid)
{
    uint8_t request = PICC_REQA;
    uint8_t atqa[2];
    uint8_t atqaLen = sizeof(atqa);

    if (!RFID_Transceive(&request, 1, atqa, &atqaLen, 0x07))
        return 0;

    uint8_t anticoll[2] = {PICC_ANTICOLL, 0x20};
    uint8_t response[5];
    uint8_t responseLen = sizeof(response);

    if (!RFID_Transceive(anticoll, 2, response,
                         &responseLen, 0x00))
        return 0;

    if (responseLen != 5)
        return 0;

    if ((uint8_t)(response[0] ^ response[1] ^
                  response[2] ^ response[3]) != response[4])
        return 0;

    for (uint8_t i = 0; i < 4; i++)
        uid[i] = response[i];

    return 1;
}

/* =========================================================
   GARAGE SERVO: PA9 / TIM1 CHANNEL 2
   Timer tick = 1 MHz, period = 20 ms (50 Hz).
   ========================================================= */
void GarageServo_Set(uint16_t pulse)
{
    __HAL_TIM_SET_COMPARE(&htim1, TIM_CHANNEL_2, pulse);
}

void GarageServo_Open(void)
{
    GarageServo_Set(GARAGE_SERVO_OPEN);

    DisplayStatus("GARAGE OPEN", "WELCOME");
    HAL_Delay(GARAGE_OPEN_TIME);

    GarageServo_Set(GARAGE_SERVO_CLOSE);

    DisplayStatus("GARAGE CLOSED", "SCAN KEYCHAIN");
    HAL_Delay(1000);
}

/* =========================================================
   MAIN
   ========================================================= */
int main(void)
{
    HAL_Init();
    SystemClock_Config();

    MX_GPIO_Init();
    MX_I2C1_Init();
    MX_SPI1_Init();
    MX_TIM1_Init();

    OLED_Init();

    if (HAL_TIM_PWM_Start(&htim1, TIM_CHANNEL_2) != HAL_OK)
        Error_Handler();

    GarageServo_Set(GARAGE_SERVO_CLOSE);

    DisplayStatus("SMART HOME", "RFID STARTING");
    HAL_Delay(1000);

    RFID_Init();

    uint8_t version = RFID_ReadRegister(REG_VERSION);

    if (version == 0x00 || version == 0xFF)
    {
        DisplayStatus("RC522 ERROR", "CHECK SPI WIRING");

        while (1)
            HAL_Delay(500);
    }

    DisplayStatus("RFID READY", "SCAN KEYCHAIN");

    uint8_t uid[4];
    uint8_t previousUID[4] = {0, 0, 0, 0};
    uint8_t havePreviousUID = 0;

    while (1)
    {
        if (RFID_ReadUID(uid))
        {
            uint8_t sameUID =
                havePreviousUID && memcmp(uid, previousUID, 4) == 0;

            if (!sameUID)
            {
                memcpy(previousUID, uid, 4);
                havePreviousUID = 1;

                if (uid[0] == RFID_KEY_UID_0 &&
                    uid[1] == RFID_KEY_UID_1 &&
                    uid[2] == RFID_KEY_UID_2 &&
                    uid[3] == RFID_KEY_UID_3)
                {
                    GarageServo_Open();
                }
                else if (uid[0] == RFID_CARD_UID_0 &&
                         uid[1] == RFID_CARD_UID_1 &&
                         uid[2] == RFID_CARD_UID_2 &&
                         uid[3] == RFID_CARD_UID_3)
                {
                    DisplayStatus("NOT AUTHORIZED", "ACCESS DENIED");
                    HAL_Delay(1500);
                    DisplayStatus("RFID READY", "SCAN KEYCHAIN");
                }
                else
                {
                    DisplayStatus("UNKNOWN TAG", "ACCESS DENIED");
                    HAL_Delay(1500);
                    DisplayStatus("RFID READY", "SCAN KEYCHAIN");
                }
            }
        }
        else
        {
            havePreviousUID = 0;
        }

        HAL_Delay(100);
    }
}

/* =========================================================
   GPIO INITIALIZATION
   ========================================================= */
static void MX_GPIO_Init(void)
{
    GPIO_InitTypeDef GPIO_InitStruct = {0};

    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOB_CLK_ENABLE();

    HAL_GPIO_WritePin(RFID_CS_PORT, RFID_CS_PIN, GPIO_PIN_SET);
    HAL_GPIO_WritePin(RFID_RST_PORT, RFID_RST_PIN, GPIO_PIN_SET);

    /* RC522 CS: PA4 */
    GPIO_InitStruct.Pin = RFID_CS_PIN;
    GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_VERY_HIGH;
    HAL_GPIO_Init(RFID_CS_PORT, &GPIO_InitStruct);

    /* RC522 reset: PB10 */
    GPIO_InitStruct.Pin = RFID_RST_PIN;
    GPIO_InitStruct.Mode = GPIO_MODE_OUTPUT_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;
    HAL_GPIO_Init(RFID_RST_PORT, &GPIO_InitStruct);

    /* OLED I2C1: PB6=SCL, PB7=SDA */
    GPIO_InitStruct.Pin = GPIO_PIN_6 | GPIO_PIN_7;
    GPIO_InitStruct.Mode = GPIO_MODE_AF_OD;
    GPIO_InitStruct.Pull = GPIO_PULLUP;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_VERY_HIGH;
    GPIO_InitStruct.Alternate = GPIO_AF4_I2C1;
    HAL_GPIO_Init(GPIOB, &GPIO_InitStruct);

    /* RC522 SPI1: PA5=SCK, PA6=MISO, PA7=MOSI */
    GPIO_InitStruct.Pin = GPIO_PIN_5 | GPIO_PIN_6 | GPIO_PIN_7;
    GPIO_InitStruct.Mode = GPIO_MODE_AF_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_VERY_HIGH;
    GPIO_InitStruct.Alternate = GPIO_AF5_SPI1;
    HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);

    /* Garage servo: PA9 / TIM1_CH2 */
    GPIO_InitStruct.Pin = GPIO_PIN_9;
    GPIO_InitStruct.Mode = GPIO_MODE_AF_PP;
    GPIO_InitStruct.Pull = GPIO_NOPULL;
    GPIO_InitStruct.Speed = GPIO_SPEED_FREQ_LOW;
    GPIO_InitStruct.Alternate = GPIO_AF1_TIM1;
    HAL_GPIO_Init(GPIOA, &GPIO_InitStruct);
}

/* =========================================================
   TIM1 PWM: GARAGE SERVO ON PA9 / CHANNEL 2
   Assumes timer clock is 16 MHz.
   ========================================================= */
static void MX_TIM1_Init(void)
{
    TIM_OC_InitTypeDef sConfigOC = {0};

    __HAL_RCC_TIM1_CLK_ENABLE();

    htim1.Instance = TIM1;
    htim1.Init.Prescaler = 15;
    htim1.Init.CounterMode = TIM_COUNTERMODE_UP;
    htim1.Init.Period = 19999;
    htim1.Init.ClockDivision = TIM_CLOCKDIVISION_DIV1;
    htim1.Init.RepetitionCounter = 0;
    htim1.Init.AutoReloadPreload = TIM_AUTORELOAD_PRELOAD_DISABLE;

    if (HAL_TIM_PWM_Init(&htim1) != HAL_OK)
        Error_Handler();

    sConfigOC.OCMode = TIM_OCMODE_PWM1;
    sConfigOC.Pulse = GARAGE_SERVO_CLOSE;
    sConfigOC.OCPolarity = TIM_OCPOLARITY_HIGH;
    sConfigOC.OCFastMode = TIM_OCFAST_DISABLE;

    if (HAL_TIM_PWM_ConfigChannel(&htim1, &sConfigOC,
                                  TIM_CHANNEL_2) != HAL_OK)
        Error_Handler();
}

/* =========================================================
   SYSTEM CLOCK: HSI 16 MHz, PLL OFF
   ========================================================= */
void SystemClock_Config(void)
{
    RCC_OscInitTypeDef RCC_OscInitStruct = {0};
    RCC_ClkInitTypeDef RCC_ClkInitStruct = {0};

    __HAL_RCC_PWR_CLK_ENABLE();
    __HAL_PWR_VOLTAGESCALING_CONFIG(PWR_REGULATOR_VOLTAGE_SCALE2);

    RCC_OscInitStruct.OscillatorType = RCC_OSCILLATORTYPE_HSI;
    RCC_OscInitStruct.HSIState = RCC_HSI_ON;
    RCC_OscInitStruct.HSICalibrationValue = RCC_HSICALIBRATION_DEFAULT;
    RCC_OscInitStruct.PLL.PLLState = RCC_PLL_NONE;

    if (HAL_RCC_OscConfig(&RCC_OscInitStruct) != HAL_OK)
        Error_Handler();

    RCC_ClkInitStruct.ClockType = RCC_CLOCKTYPE_HCLK |
                                  RCC_CLOCKTYPE_SYSCLK |
                                  RCC_CLOCKTYPE_PCLK1 |
                                  RCC_CLOCKTYPE_PCLK2;

    RCC_ClkInitStruct.SYSCLKSource = RCC_SYSCLKSOURCE_HSI;
    RCC_ClkInitStruct.AHBCLKDivider = RCC_SYSCLK_DIV1;
    RCC_ClkInitStruct.APB1CLKDivider = RCC_HCLK_DIV1;
    RCC_ClkInitStruct.APB2CLKDivider = RCC_HCLK_DIV1;

    if (HAL_RCC_ClockConfig(&RCC_ClkInitStruct,
                            FLASH_LATENCY_0) != HAL_OK)
        Error_Handler();
}

/* =========================================================
   ERROR HANDLER
   ========================================================= */
void Error_Handler(void)
{
    __disable_irq();

    while (1)
    {
    }
}
