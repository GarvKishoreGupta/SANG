/* =========================================================
   STM32F411CEU6 SMART HOME PASSWORD LOCK

   OLED: SH1106-compatible, I2C1 PB6/PB7, address 0x3C

   Keypad rows:
   R1 = PB0, R2 = PB1, R3 = PB2, R4 = PB3

   Keypad columns:
   C1 = PA15, C2 = PB12, C3 = PB13

   SG90 Servo:
   Signal = PA8 (TIM1_CH1)
   ========================================================= */

#define OLED_ADDR (0x3C << 1)

#define PASSWORD "1234"
#define MAX_PASSWORD_LENGTH 8

#define SERVO_LOCKED_PULSE    1000U
#define SERVO_UNLOCKED_PULSE  2000U

    {'4', '5', '6'},
    {'7', '8', '9'},
    {'*', '0', '#'}
};

/* =========================================================
   OLED DRIVER
   ========================================================= */

static void OLED_Command(uint8_t cmd)
{
    uint8_t buf[2] = {0x00, cmd};

    HAL_I2C_Master_Transmit(
        &hi2c1, OLED_ADDR, buf, 2, 100
    );
}

static void OLED_Data(uint8_t data)
{
    uint8_t buf[2] = {0x40, data};

    HAL_I2C_Master_Transmit(
        &hi2c1, OLED_ADDR, buf, 2, 100
    );
}

static void OLED_Init(void)
{
    HAL_Delay(100);

    OLED_Command(0xAE);
    OLED_Command(0xD5); OLED_Command(0x80);
    OLED_Command(0xA8); OLED_Command(0x3F);
    OLED_Command(0xD3); OLED_Command(0x00);
    OLED_Command(0x40);
    OLED_Command(0xAD); OLED_Command(0x8B);
    OLED_Command(0xA1);
    OLED_Command(0xC8);
    OLED_Command(0xDA); OLED_Command(0x12);
    OLED_Command(0x81); OLED_Command(0x7F);
    OLED_Command(0xD9); OLED_Command(0x22);
    OLED_Command(0xDB); OLED_Command(0x35);
    OLED_Command(0xA4);
    OLED_Command(0xA6);
    OLED_Command(0xAF);
}

static void OLED_SetPosition(uint8_t page, uint8_t col)
{
    col += 2;

    OLED_Command(0xB0 | page);
    OLED_Command(0x10 | (col >> 4));
    OLED_Command(0x00 | (col & 0x0F));
}

static void OLED_Clear(void)
{
    for (uint8_t page = 0; page < 8; page++)
    {
        OLED_SetPosition(page, 0);

        for (uint8_t col = 0; col < 128; col++)
            OLED_Data(0x00);
    }
}

static void OLED_Char(char c)
{
    if (c >= 'a' && c <= 'z')
        c -= ('a' - 'A');

    if (c < 32 || c > 90)
        c = ' ';

    uint8_t index = (uint8_t)c - 32;

    for (uint8_t i = 0; i < 5; i++)
        OLED_Data(font5x7[index][i]);

    OLED_Data(0x00);
}

static void OLED_String(const char *str)
{
    while (*str != '\0')
        OLED_Char(*str++);
}

/* =========================================================
   KEYPAD INITIALIZATION
   ========================================================= */

static void Keypad_Init(void)
{
    GPIO_InitTypeDef gpio = {0};

    __HAL_RCC_GPIOA_CLK_ENABLE();
    __HAL_RCC_GPIOB_CLK_ENABLE();

    /* Rows PB0-PB3: output push-pull */
    gpio.Pin = GPIO_PIN_0 | GPIO_PIN_1 |
               GPIO_PIN_2 | GPIO_PIN_3;
    gpio.Mode = GPIO_MODE_OUTPUT_PP;
    gpio.Pull = GPIO_NOPULL;
    gpio.Speed = GPIO_SPEED_FREQ_LOW;

    HAL_GPIO_Init(GPIOB, &gpio);

    HAL_GPIO_WritePin(
        GPIOB,
        GPIO_PIN_0 | GPIO_PIN_1 |
        GPIO_PIN_2 | GPIO_PIN_3,
        GPIO_PIN_SET
    );

    /* Column C1: PA15 */
    gpio.Pin = GPIO_PIN_15;
    gpio.Mode = GPIO_MODE_INPUT;
    gpio.Pull = GPIO_PULLUP;
    HAL_GPIO_Init(GPIOA, &gpio);

    /* Columns C2 and C3: PB12 and PB13 */
    gpio.Pin = GPIO_PIN_12 | GPIO_PIN_13;
    gpio.Mode = GPIO_MODE_INPUT;
    gpio.Pull = GPIO_PULLUP;
    HAL_GPIO_Init(GPIOB, &gpio);
}

/* =========================================================
   KEYPAD SCANNER
   ========================================================= */

static char Keypad_GetKey(void)
{
    for (int row = 0; row < 4; row++)
    {
        /* All rows HIGH */
        for (int r = 0; r < 4; r++)
        {
            HAL_GPIO_WritePin(
                rowPort[r], rowPin[r], GPIO_PIN_SET
            );
        }

        /* Current row LOW */
        HAL_GPIO_WritePin(
            rowPort[row], rowPin[row], GPIO_PIN_RESET
        );

        HAL_Delay(2);

        for (int col = 0; col < 3; col++)
        {
            if (HAL_GPIO_ReadPin(
                    colPort[col], colPin[col]
                ) == GPIO_PIN_RESET)
            {
                HAL_Delay(20);

                if (HAL_GPIO_ReadPin(
                        colPort[col], colPin[col]
                    ) == GPIO_PIN_RESET)
                {
                    char key = keyMap[row][col];

                    /* Wait until key is released */
                    while (HAL_GPIO_ReadPin(
                               colPort[col], colPin[col]
                           ) == GPIO_PIN_RESET)
                    {
                        HAL_Delay(5);
                    }

                    HAL_Delay(30);
                    return key;
                }
            }
        }
    }

    return '\0';
}

/* =========================================================
   SG90 SERVO - PA8 / TIM1_CH1
   Timer clock: 16 MHz
   Counter clock: 1 MHz
   PWM frequency: 50 Hz
   ========================================================= */

static void Servo_SetPosition(uint16_t pulse)
{
    __HAL_TIM_SET_COMPARE(
        &htim1, TIM_CHANNEL_1, pulse
    );
}

static void MX_TIM1_Init(void)
{
    TIM_ClockConfigTypeDef sClockSourceConfig = {0};
    TIM_OC_InitTypeDef sConfigOC = {0};
    TIM_MasterConfigTypeDef sMasterConfig = {0};

    __HAL_RCC_TIM1_CLK_ENABLE();
    __HAL_RCC_GPIOA_CLK_ENABLE();

    htim1.Instance = TIM1;
    htim1.Init.Prescaler = 15;
    htim1.Init.CounterMode = TIM_COUNTERMODE_UP;
    htim1.Init.Period = 19999;
    htim1.Init.ClockDivision = TIM_CLOCKDIVISION_DIV1;
    htim1.Init.RepetitionCounter = 0;
    htim1.Init.AutoReloadPreload =
        TIM_AUTORELOAD_PRELOAD_DISABLE;

    if (HAL_TIM_Base_Init(&htim1) != HAL_OK)
        Error_Handler();

    sClockSourceConfig.ClockSource =
        TIM_CLOCKSOURCE_INTERNAL;

    if (HAL_TIM_ConfigClockSource(
            &htim1, &sClockSourceConfig) != HAL_OK)
        Error_Handler();

    if (HAL_TIM_PWM_Init(&htim1) != HAL_OK)
        Error_Handler();

    sConfigOC.OCMode = TIM_OCMODE_PWM1;
    sConfigOC.Pulse = SERVO_LOCKED_PULSE;
    sConfigOC.OCPolarity = TIM_OCPOLARITY_HIGH;
    sConfigOC.OCFastMode = TIM_OCFAST_DISABLE;

    if (HAL_TIM_PWM_ConfigChannel(
            &htim1, &sConfigOC, TIM_CHANNEL_1) != HAL_OK)
        Error_Handler();

    sMasterConfig.MasterOutputTrigger = TIM_TRGO_RESET;
    sMasterConfig.MasterSlaveMode =
        TIM_MASTERSLAVEMODE_DISABLE;

    if (HAL_TIMEx_MasterConfigSynchronization(
            &htim1, &sMasterConfig) != HAL_OK)
        Error_Handler();

    /* PA8 = TIM1_CH1 */
    GPIO_InitTypeDef gpio = {0};

    gpio.Pin = GPIO_PIN_8;
    gpio.Mode = GPIO_MODE_AF_PP;
    gpio.Pull = GPIO_NOPULL;
    gpio.Speed = GPIO_SPEED_FREQ_LOW;
    gpio.Alternate = GPIO_AF1_TIM1;

    HAL_GPIO_Init(GPIOA, &gpio);
}

/* =========================================================
   PASSWORD DISPLAY
   ========================================================= */

static void DisplayPasswordScreen(
    const char *maskedPassword,
    const char *message
)
{
    OLED_Clear();

    OLED_SetPosition(0, 10);
    OLED_String("SMART HOME");

    OLED_SetPosition(2, 0);
    OLED_String("ENTER PASSWORD:");

    OLED_SetPosition(4, 0);

    if (maskedPassword[0] == '\0')
        OLED_String("_");
    else
        OLED_String(maskedPassword);

    OLED_SetPosition(6, 0);
    OLED_String(message);
}

/* =========================================================
   MAIN APPLICATION
   ========================================================= */

int main(void)
{
    HAL_Init();
    SystemClock_Config();

    MX_GPIO_Init();
    MX_I2C1_Init();
    Keypad_Init();

    /* Initialize and start servo PWM */
    MX_TIM1_Init();

    if (HAL_TIM_PWM_Start(
            &htim1, TIM_CHANNEL_1) != HAL_OK)
    {
        Error_Handler();
    }

    /* Set locked position */
    Servo_SetPosition(SERVO_LOCKED_PULSE);
    HAL_Delay(500);

    OLED_Init();
    OLED_Clear();

    OLED_SetPosition(2, 10);
    OLED_String("SMART HOME");

    OLED_SetPosition(4, 10);
    OLED_String("PASSWORD LOCK");

    HAL_Delay(1500);

    char enteredPassword[MAX_PASSWORD_LENGTH + 1] = {0};
    char maskedPassword[MAX_PASSWORD_LENGTH + 1] = {0};

    uint8_t passwordLength = 0;
    const char *message = "USE # TO ENTER";

    DisplayPasswordScreen(maskedPassword, message);

    while (1)
    {
        char key = Keypad_GetKey();

        if (key == '\0')
            continue;

        /* Clear entered password */
        if (key == '*')
        {
            passwordLength = 0;

            memset(enteredPassword, 0,
                   sizeof(enteredPassword));

            memset(maskedPassword, 0,
                   sizeof(maskedPassword));

            message = "USE # TO ENTER";

            DisplayPasswordScreen(
                maskedPassword, message
            );
        }

        /* Submit password */
        else if (key == '#')
        {
            enteredPassword[passwordLength] = '\0';

            if (strcmp(enteredPassword, PASSWORD) == 0)
            {
                message = "ACCESS GRANTED";

                DisplayPasswordScreen(
                    maskedPassword, message
                );

                /* Unlock servo */
                Servo_SetPosition(
                    SERVO_UNLOCKED_PULSE
                );

                /* Stay unlocked for 3 seconds */
                HAL_Delay(3000);

                /* Lock servo again */
                Servo_SetPosition(
                    SERVO_LOCKED_PULSE
                );

                HAL_Delay(1000);
            }
            else
            {
                message = "ACCESS DENIED";

                DisplayPasswordScreen(
                    maskedPassword, message
                );

                HAL_Delay(2000);
            }

            /* Reset for next attempt */
            passwordLength = 0;

            memset(enteredPassword, 0,
                   sizeof(enteredPassword));

            memset(maskedPassword, 0,
                   sizeof(maskedPassword));

            message = "USE # TO ENTER";

            DisplayPasswordScreen(
                maskedPassword, message
            );
        }

        /* Accept numeric digits */
        else if (key >= '0' && key <= '9')
        {
            if (passwordLength < MAX_PASSWORD_LENGTH)
            {
                enteredPassword[passwordLength] = key;
                passwordLength++;

                enteredPassword[passwordLength] = '\0';

                maskedPassword[passwordLength - 1] = '*';
                maskedPassword[passwordLength] = '\0';

                message = "USE # TO ENTER";

                DisplayPasswordScreen(
                    maskedPassword, message
                );
            }
        }
    }
}

/* =========================================================
   CLOCK CONFIGURATION
   HSI 16 MHz, PLL disabled
   ========================================================= */

void SystemClock_Config(void)
{
    RCC_OscInitTypeDef osc = {0};
    RCC_ClkInitTypeDef clk = {0};

    __HAL_RCC_PWR_CLK_ENABLE();

    __HAL_PWR_VOLTAGESCALING_CONFIG(
        PWR_REGULATOR_VOLTAGE_SCALE2
    );

    osc.OscillatorType = RCC_OSCILLATORTYPE_HSI;
    osc.HSIState = RCC_HSI_ON;
    osc.HSICalibrationValue = RCC_HSICALIBRATION_DEFAULT;
    osc.PLL.PLLState = RCC_PLL_NONE;

    if (HAL_RCC_OscConfig(&osc) != HAL_OK)
        Error_Handler();

    clk.ClockType = RCC_CLOCKTYPE_HCLK |
                    RCC_CLOCKTYPE_SYSCLK |
                    RCC_CLOCKTYPE_PCLK1 |
                    RCC_CLOCKTYPE_PCLK2;

    clk.SYSCLKSource = RCC_SYSCLKSOURCE_HSI;
    clk.AHBCLKDivider = RCC_SYSCLK_DIV1;
    clk.APB1CLKDivider = RCC_HCLK_DIV1;
    clk.APB2CLKDivider = RCC_HCLK_DIV1;

    if (HAL_RCC_ClockConfig(
            &clk, FLASH_LATENCY_0) != HAL_OK)
    {
        Error_Handler();
    }
}

/* =========================================================
   GPIO CONFIGURATION
   ========================================================= */

static void MX_GPIO_Init(void)
{
    GPIO_InitTypeDef gpio = {0};

    __HAL_RCC_GPIOB_CLK_ENABLE();

    /* OLED I2C1 pins PB6/PB7 */
    gpio.Pin = GPIO_PIN_6 | GPIO_PIN_7;
    gpio.Mode = GPIO_MODE_AF_OD;
    gpio.Pull = GPIO_PULLUP;
    gpio.Speed = GPIO_SPEED_FREQ_VERY_HIGH;
    gpio.Alternate = GPIO_AF4_I2C1;

    HAL_GPIO_Init(GPIOB, &gpio);
}

/* =========================================================
   I2C1 CONFIGURATION
   ========================================================= */

static void MX_I2C1_Init(void)
{
    __HAL_RCC_I2C1_CLK_ENABLE();

    hi2c1.Instance = I2C1;
    hi2c1.Init.ClockSpeed = 100000;
    hi2c1.Init.DutyCycle = I2C_DUTYCYCLE_2;
    hi2c1.Init.OwnAddress1 = 0;
    hi2c1.Init.AddressingMode = I2C_ADDRESSINGMODE_7BIT;
    hi2c1.Init.DualAddressMode = I2C_DUALADDRESS_DISABLE;
    hi2c1.Init.OwnAddress2 = 0;
    hi2c1.Init.GeneralCallMode = I2C_GENERALCALL_DISABLE;
    hi2c1.Init.NoStretchMode = I2C_NOSTRETCH_DISABLE;

    if (HAL_I2C_Init(&hi2c1) != HAL_OK)
        Error_Handler();
}

   ERROR HANDLER

{
    __disable_irq();

    {
    }
